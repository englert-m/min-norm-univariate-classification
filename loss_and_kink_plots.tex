\newcommand{\rangeSeries}[5]{
    \addplot[draw=none,name path=#5-upper,forget plot] table[col sep=comma,x=num_class_breaks,y=max_#2] {#1};
    \addplot[draw=none,name path=#5-lower,forget plot] table[col sep=comma,x=num_class_breaks,y=min_#2] {#1};
    \addplot[draw=none,fill=#3,fill opacity=0.18,forget plot] fill between[of=#5-upper and #5-lower];
    \addplot[#3,#4] table[col sep=comma,x=num_class_breaks,y=mean_#2] {#1};
}
